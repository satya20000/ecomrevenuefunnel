// Add each measure individually. Percentages are fractions; format 0.0%.
Total MQLs = COUNTROWS(fact_lead)
Closed Deals = SUM(fact_lead[converted])
Conversion Rate = DIVIDE([Closed Deals], [Total MQLs])
Active Sellers = SUM(fact_lead[active])
Activation Rate = DIVIDE([Active Sellers], [Closed Deals])
Attributed GMV = SUM(fact_lead[gmv])
GMV per Lead = DIVIDE([Attributed GMV], [Total MQLs])
GMV per Converted Seller = DIVIDE([Attributed GMV], [Closed Deals])
Average Close Days = AVERAGE(fact_lead[close_days])
Average First Sale Days = AVERAGE(fact_lead[close_to_first_sale_days])
Repeat Sellers = SUM(fact_lead[repeat_seller])
Repeat Seller Rate = DIVIDE([Repeat Sellers], [Active Sellers])
Seller Order Pairs = SUM(fact_lead[orders])
Orders per Active Seller = DIVIDE([Seller Order Pairs], [Active Sellers])
Attributed Marketplace Orders = DISTINCTCOUNT(fact_seller_order[order_id])
Attributed Item AOV = DIVIDE(SUM(fact_seller_order[gmv]), [Attributed Marketplace Orders])
Top 10 Percent GMV Share =
 VAR Sellers = FILTER(ALLSELECTED(fact_lead), fact_lead[active] = 1)
 VAR TopNCount = ROUNDUP(COUNTROWS(Sellers) * 0.1, 0)
 VAR TopSellers = TOPN(TopNCount, Sellers, fact_lead[gmv], DESC, fact_lead[seller_id], ASC)
 RETURN DIVIDE(SUMX(TopSellers, fact_lead[gmv]), SUMX(Sellers, fact_lead[gmv]))
